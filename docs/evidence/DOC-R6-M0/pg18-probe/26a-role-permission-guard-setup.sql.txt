-- as lawdesk_owner: layer 5 of 05 A.2 on the REAL role_permission table. The guard applies to every role, the owner included.
create table lawdesk.sensitive_permission (code varchar(64) primary key)
;;
insert into lawdesk.sensitive_permission values ('POSTJ_APPROVE'), ('APPEAL_SETTINGS_APPROVE'), ('LEGAL_GATE_APPROVE'), ('LEGAL_AUTHORITY_CUSTODIAN'), ('COMPLAINT_VIEW'), ('COMPLAINT_CREATE'), ('COMPLAINT_EDIT'), ('COMPLAINT_PREPARE_DECISION'), ('COMPLAINT_APPROVE')
;;
revoke insert, update, delete, truncate on lawdesk.sensitive_permission from lawdesk_dml
;;
create function lawdesk.r6_guard_role_permission() returns trigger language plpgsql set search_path = pg_catalog, lawdesk, pg_temp as $$
begin
  if exists (select 1 from lawdesk.sensitive_permission s where s.code = new.permission) then
    raise exception 'permission % is a sensitive legal authority and cannot be attached to a role', new.permission using errcode = 'LD060';
  end if;
  return new;
end $$
;;
create trigger r6_guard_role_permission before insert or update on lawdesk.role_permission for each row execute function lawdesk.r6_guard_role_permission()
;;
-- the owner itself is refused too
insert into lawdesk.role_permission (role_id, permission) select id, 'LEGAL_AUTHORITY_CUSTODIAN' from lawdesk.app_role where code = 'ADMIN'
;;
