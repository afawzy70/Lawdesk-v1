create function lawdesk.r6_check_referral_invariant() returns trigger language plpgsql security definer set search_path = pg_catalog, lawdesk, pg_temp as $$
declare cid uuid; st text; n int;
begin
  if tg_table_name = 'complaint' then cid := coalesce(new.id, old.id); else cid := coalesce(new.complaint_id, old.complaint_id); end if;
  select status into st from lawdesk.complaint where id = cid;
  select count(*) into n from lawdesk.complaint_case_link where complaint_id = cid;
  if st is null then return null; end if;
  if (st = 'REFERRED') <> (n = 1) or n > 1 then
    raise exception 'invariant R-1 violated for complaint %: status=% links=%', cid, st, n using errcode = 'LD032'; end if;
  return null;
end $$
;;
create constraint trigger r6_referral_invariant_c after insert or update of status on lawdesk.complaint deferrable initially deferred for each row execute function lawdesk.r6_check_referral_invariant()
;;
create constraint trigger r6_referral_invariant_l after insert or update or delete on lawdesk.complaint_case_link deferrable initially deferred for each row execute function lawdesk.r6_check_referral_invariant()
;;
create function lawdesk.r6_complete_referral(p_complaint uuid, p_case uuid, p_actor uuid) returns void language plpgsql security definer set search_path = pg_catalog, lawdesk, pg_temp as $$
declare st text; cs lawdesk.legal_case%rowtype; d lawdesk.referral_draft%rowtype;
begin
  select status into st from lawdesk.complaint where id = p_complaint for update;
  if st is null then raise exception 'complaint not found' using errcode = 'LD020'; end if;
  if st = 'REFERRED' then raise exception 'complaint already referred' using errcode = 'LD021'; end if;
  if st <> 'UNDER_INVESTIGATION' then raise exception 'complaint status % cannot be referred', st using errcode = 'LD022'; end if;
  if not exists (select 1 from lawdesk.owner_policy where policy = 'P-MANIFEST' and enabled) then raise exception 'policy P-MANIFEST is off' using errcode = 'LD023'; end if;
  select * into d from lawdesk.referral_draft where complaint_id = p_complaint;
  if not found or not d.manifest_approved or cardinality(d.missing) > 0 then raise exception 'referral draft is incomplete or unapproved' using errcode = 'LD024'; end if;
  if not exists (select 1 from lawdesk.legal_authority_grant g where g.user_id = p_actor and g.authority = 'COMPLAINT_APPROVE' and g.state = 'ACTIVE') then raise exception 'actor lacks COMPLAINT_APPROVE' using errcode = 'LD005'; end if;
  select * into cs from lawdesk.legal_case where id = p_case for update;
  if not found or cs.confidentiality <> 'RESTRICTED' or cs.created_at <> transaction_timestamp() then
    raise exception 'case must be inserted RESTRICTED in this very transaction' using errcode = 'LD025'; end if;
  insert into lawdesk.complaint_case_link (complaint_id, case_id) values (p_complaint, p_case);
  update lawdesk.complaint set status = 'REFERRED' where id = p_complaint;
end $$
;;
revoke all on function lawdesk.r6_complete_referral(uuid, uuid, uuid) from public
